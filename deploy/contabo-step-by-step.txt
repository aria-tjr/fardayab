10. Create simple bot:
mkdir -p /opt/telegram-bot/telegram_bot
cat > /opt/telegram-bot/telegram_bot/main.py << 'EOF'
import asyncio
import os
from aiogram import Bot, Dispatcher
from aiogram.enums import ParseMode
from aiogram.client.default import DefaultBotProperties

async def main():
    bot = Bot(token=os.getenv("TELEGRAM_BOT_TOKEN", "your_bot_token_here"), 
              default=DefaultBotProperties(parse_mode=ParseMode.HTML))
    dp = Dispatcher()
    
    @dp.message()
    async def handle_message(message):
        await message.answer(f"🤖 Bot is running on Contabo IPv6: 2a02:c207:3017:9335::1")
    
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
EOF

11. Create requirements.txt:
cat > /opt/telegram-bot/telegram_bot/requirements.txt << 'EOF'
aiogram==3.4.1
python-dotenv==1.0.0
EOF

12. Create Dockerfile:
cat > /opt/telegram-bot/telegram_bot/Dockerfile << 'EOF'
FROM python:3.12-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install -r requirements.txt
COPY . .
CMD ["python", "main.py"]
EOF

13. Create Docker Compose:
cat > /opt/telegram-bot/deploy/docker-compose.yml << 'EOF'
version: "3.9"
services:
  telegram-bot:
    build: ../telegram_bot
    container_name: telegram-bot
    env_file: ../config/.env
    restart: unless-stopped
    networks:
      - bot-network
    ports:
      - "2a02:c207:3017:9335::1:8080:8080"

  nginx:
    image: nginx:alpine
    container_name: telegram-bot-nginx
    ports:
      - "2a02:c207:3017:9335::1:80:80"
      - "2a02:c207:3017:9335::1:443:443"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
      - ../ssl:/etc/nginx/ssl:ro
    depends_on:
      - telegram-bot
    restart: unless-stopped
    networks:
      - bot-network

networks:
  bot-network:
    driver: bridge
    enable_ipv6: true
    ipam:
      config:
        - subnet: 172.20.0.0/16
        - subnet: 2a02:c207:3017:9335::/64
EOF

14. Create Nginx config:
cat > /opt/telegram-bot/deploy/nginx.conf << 'EOF'
events { worker_connections 1024; }
http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;
    
    server {
        listen [2a02:c207:3017:9335::1]:80;
        server_name _;
        return 301 https://$server_name$request_uri;
    }
    
    server {
        listen [2a02:c207:3017:9335::1]:443 ssl http2;
        server_name _;
        
        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        
        location / {
            proxy_pass http://telegram-bot:8080;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
        
        location /health {
            return 200 "healthy\n";
            add_header Content-Type text/plain;
        }
    }
}
EOF
